import 'fake-indexeddb/auto';
import { describe, it, expect, vi } from 'vitest';
import { createSeededRandom } from '../../utils/random';
import { createInitialProfile } from '../user/profile';
import { generateMathProblem, MATH_GENERATORS } from '../math';
import { buildPeriodicTestSet } from '../test/testSet';
import { prepareStudyBlockPresentation, resolveStudyHissanPresentation } from '../math/studyPresentation';
import { studyLearningEvidence, learningEvidenceForProblem } from './attemptContext';
import { integerFractionProblem } from '../math/fractionInput';
import { finishEligibility } from '../finishTest';
import { getWordsByLevel } from '../english/words';
import { getSkillsForLevel } from '../math/curriculum';

import { shouldForceVocabReviewBlock } from '../../hooks/blockGenerators';
import { resolveWeakStateAfterAttempt } from '../learningAttemptWriter';
import { getMathFollowupPlan } from '../math/followups';
import { planMathProblems } from '../math/planner';
import { updateMemoryState } from '../algorithms/srs';
import { db } from '../../db';
import { saveProfile, getAppData, saveAppData, getProfile } from '../user/repository';
import { recordFinishTestAnswer, reserveFinishTest, completeFinishTest } from '../finishTestRepository';
function profile(level=8) {const p=createInitialProfile('audit',2,level,1,'math');p.mathMainLevel=level;p.mathMaxUnlocked=level;return p;}
function p(skill:string):Problem {return {...generateMathProblem(skill,{random:skill==='frac_add_same'?()=>0:createSeededRandom('audit')}),id:'diagnostic',subject:'math',isReview:false};}
describe('read-only policy diagnostics (assertions describe observed behavior, not acceptance)',()=>{
 it('D1: default written presentation loses finish provenance',()=>{const raw=p('add_2d2d_c');const [frozen]=prepareStudyBlockPresentation([raw],true,true);const shown=resolveStudyHissanPresentation(frozen,true);expect(shown.isHissanActive).toBe(true);expect(studyLearningEvidence(frozen,'independent',true,false)).toBeUndefined();console.log('D1',JSON.stringify({skill:raw.categoryId,input:frozen.inputType,written:shown.isHissanActive,evidence:false}));});
 it('D2: whole fraction presentation loses normal and finish provenance',()=>{const raw=p('frac_add_same');expect(raw.correctAnswer).toEqual(['1','1']);const displayed=integerFractionProblem(raw);expect(displayed.inputType).toBe('number');expect(studyLearningEvidence(displayed,'independent',false,false)).toBeUndefined();console.log('D2',JSON.stringify({original:raw.correctAnswer,displayed:displayed.correctAnswer,evidence:false}));});
 it('D3: Lv9 with ten independent successes per skill still tests one non-carry equation',()=>{const spy=vi.spyOn(Math,'random').mockReturnValue(0);try{const q=profile(9);for(const id of getSkillsForLevel(9))q.mathSkills[id]={...createDefaultMemoryState(id,'math',true),totalAnswers:10,correctAnswers:10,independentCorrectAnswers:10};q.mathLevels=q.mathLevels?.map(l=>l.level===9?{...l,recentIndependentAnswersNonReview:Array(20).fill(true)}:l);expect(finishEligibility(q,'math').status).toBe('ready');const set=buildPeriodicTestSet(q,'math');const texts=new Set(set.problems.map(q=>q.questionText));expect(set.problems.length).toBe(20);expect(texts.size).toBe(1);console.log('D3',JSON.stringify({questions:20,equations:[...texts],skills:[...new Set(set.problems.map(q=>q.categoryId))]}));}finally{spy.mockRestore();}});
 it('D4: readiness ignores elapsed time and distinct items',()=>{const q=profile();q.mathLevels=q.mathLevels?.map(l=>l.level===8?{...l,recentIndependentAnswersNonReview:Array(20).fill(true),updatedAt:'2020-01-01T00:00:00Z'}:l);expect(finishEligibility(q,'math').status).toBe('ready');console.log('D4',{status:'ready',updatedAt:'2020-01-01T00:00:00Z'});});
 it('D5: vocab eligibility has no word coverage requirement',()=>{const q=profile();q.subjectMode='vocab';q.vocabMainLevel=1;q.vocabMaxUnlocked=1;q.vocabLevels=q.vocabLevels?.map(l=>l.level===1?{...l,recentIndependentAnswersNonReview:Array(20).fill(true)}:l);expect(Object.keys(q.vocabWords).length).toBe(0);expect(finishEligibility(q,'vocab').status).toBe('ready');console.log('D5',{eligibleWords:getWordsByLevel(1).length,recordedWords:0,status:'ready'});});
 it('D6: a single Due forces English even after 40 English questions at 10% review',()=>{const h=Array.from({length:40},(_,i)=>({subject:'vocab' as const,isReview:i%10===0}));expect(shouldForceVocabReviewBlock('mix',1,h)).toBe(true);console.log('D6',{vocabHistory:40,reviewRatio:0.1,due:1,forcedEnglish:true});});
 it('D7: raw corrected successes can clear weak without independent successes',()=>{expect(resolveWeakStateAfterAttempt(true,Array(10).fill('correct'))).toBe(false);console.log('D7',{rawCorrect:10,previousWeak:true,nextWeak:false});});
 it('D8: presentation followup trusts raw correct',()=>{const r=getMathFollowupPlan([{id:'1',subject:'math',skillId:'add_2d1d_nc_bridge',result:'correct',timestamp:'2026-10-01T00:00:00Z'}],getSkillsForLevel(11),11);expect(r.some(x=>x.reason==='progression')).toBe(true);console.log('D8',JSON.stringify(r));});
 it('D9: old lowest-strength Due can fill both slots of short math section',()=>{const q=profile(28);const id=getSkillsForLevel(28)[0];q.mathSkills[id]={...createDefaultMemoryState(id,'math',false),strength:1};const r=planMathProblems({profile:q,count:2,dueSkillIds:[id],canAddReview:(items)=>items.filter(i=>i.countsTowardReviewCap).length<Math.max(1,Math.floor((items.length+1)*0.6)),random:()=>0});expect(r.map(x=>x.source)).toEqual(['due','main']);expect(r[0].skillId).toBe(r[1].skillId);console.log('D9',JSON.stringify(r));});
 it('D10: independent retry within 24h cannot advance SRS',()=>{const m={...createDefaultMemoryState('add_1d_1','math',false),strength:3,updatedAt:'2026-09-30T00:00:00Z',lastIndependentCorrectAt:'2026-09-30T00:00:00Z',nextReview:'2026-10-01T00:00:00Z'};const n=updateMemoryState(m,true,false,new Date('2026-10-01T00:00:00Z'),{independence:'independent',wholeProblem:true,latestChallengeContactAt:'2026-09-30T23:00:00Z'});expect(n.strength).toBe(3);const ignored=updateMemoryState(m,true,false,new Date('2026-10-01T00:00:00Z'),{independence:'independent',wholeProblem:true});expect(ignored.strength).toBe(4);console.log('D10',{withContact:n.strength,unrecordedContact:ignored.strength});});
 it('D11: failed finish does not write memory, logs or contact; immediate retry remains available',async()=>{await db.delete();await db.open();const q=profile();q.mathLevels=q.mathLevels?.map(l=>l.level===8?{...l,recentIndependentAnswersNonReview:Array(20).fill(true)}:l);await saveProfile(q);const data=await getAppData();await saveAppData({...data,activeProfileId:q.id});const set=await reserveFinishTest(q.id,'math');for(let i=0;i<20;i++)await recordFinishTestAnswer(set,i,i!==0);const c=await completeFinishTest(set,{correct:19,total:20,durationSeconds:10});expect(c?.result.passed).toBe(false);expect(await db.logs.count()).toBe(0);expect(await db.memoryMath.count()).toBe(0);expect(await db.challengeContacts.count()).toBe(0);const retry=await reserveFinishTest(q.id,'math');expect(retry.id).not.toBe(set.id);console.log('D11',{logs:0,memory:0,contacts:0,retryFresh:true,main:(await getProfile(q.id))?.mathMainLevel});await db.delete();});
 it('D12: seeded smoke over every math generator; arithmetic spot invariants',()=>{let generated=0,arithmetic=0;for(const skill of Object.keys(MATH_GENERATORS)){for(let i=0;i<100;i++){const raw=generateMathProblem(skill,{random:createSeededRandom(`audit:${skill}:${i}`)});expect(raw.categoryId).toBe(skill);expect(raw.correctAnswer).toBeDefined();const match=raw.questionText?.match(/^(\d+(?:\.\d+)?)\s*([+×÷-])\s*(\d+(?:\.\d+)?)\s*=\s*$/);if(match&&typeof raw.correctAnswer==='string'&&raw.inputType==='number'){const a=Number(match[1]),b=Number(match[3]);const expected=match[2]==='+'?a+b:match[2]==='-'?a-b:match[2]==='×'?a*b:a/b;expect(Number(raw.correctAnswer)).toBeCloseTo(expected,7);arithmetic++;}generated++;}}console.log('D12',{skills:Object.keys(MATH_GENERATORS).length,generated,arithmetic});});

});
