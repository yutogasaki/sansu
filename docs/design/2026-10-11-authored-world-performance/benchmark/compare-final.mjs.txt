import fs from 'node:fs';import assert from 'node:assert/strict';
const taskRoot='/tmp/sansu-authored-equivalence-a2ea3e98',beforePrefix=process.argv[2]??taskRoot+'/baseline',afterPrefix=process.argv[3]??taskRoot+'/optimized';
const before=JSON.parse(fs.readFileSync(beforePrefix+'.results.json','utf8')),after=JSON.parse(fs.readFileSync(afterPrefix+'.results.json','utf8'));
for(const [prefix,result]of [[beforePrefix,before],[afterPrefix,after]]){const config=JSON.parse(fs.readFileSync(prefix+'.build.json','utf8'));assert.deepEqual(result.inputShaStart,result.inputShaEnd,`${prefix}: measurement SHA stable`);for(const [file,sha]of Object.entries(config.sourceHashes))assert.equal(result.inputShaStart[file],sha,`${prefix}: source hash matches bundled source ${file}`);for(const input of Object.values(config.inputs))assert.equal(result.inputShaStart[input.file],input.sha256,`${prefix}: frozen input ${input.file}`);}
const components=Object.fromEntries(Object.keys(before.fingerprints).map(name=>[name,{same:before.fingerprints[name].sha256===after.fingerprints[name]?.sha256,before:before.fingerprints[name],after:after.fingerprints[name]}]));const timings=before.events.map(event=>{const next=after.events.find(x=>x.name===event.name);return {name:event.name,beforeMs:event.ms,afterMs:next?.ms,speedup:next&&event.ms/next.ms,beforeCalls:event.counts,afterCalls:next?.counts};});
function differences(a,b,path='',out=[]){if(out.length>=30)return out;if(JSON.stringify(a)===JSON.stringify(b))return out;if(a===null||b===null||typeof a!=='object'||typeof b!=='object'){out.push({path,before:a,after:b});return out;}for(const key of new Set([...Object.keys(a),...Object.keys(b)])){differences(a[key],b[key],path+(Array.isArray(a)?`[${key}]`:`.${key}`),out);if(out.length>=30)break;}return out;}
const mismatches=Object.entries(components).filter(([,item])=>!item.same).map(([name])=>name),detail={};if(mismatches.includes('objectLayer.allNodes')){const a=JSON.parse(fs.readFileSync(beforePrefix+'.object-nodes.json','utf8')),b=JSON.parse(fs.readFileSync(afterPrefix+'.object-nodes.json','utf8'));detail.objectNodes=differences(a,b);}
if(mismatches.includes('layout.floor-point-inverse.samples')){const a=JSON.parse(fs.readFileSync(beforePrefix+'.layout-samples.json','utf8')),b=JSON.parse(fs.readFileSync(afterPrefix+'.layout-samples.json','utf8'));detail.layoutSamples=differences(a,b);}
const report={schema:'sansu-authored-performance-equivalence-comparison-v1',before:before.variant,after:after.variant,sameProcessConditions:before.node===after.node&&JSON.stringify(before.versions)===JSON.stringify(after.versions),sameFrozenInputs:true,exactOutputEquivalence:mismatches.length===0,components,mismatches,summaryBefore:before.summary,summaryAfter:after.summary,timings,detail};fs.writeFileSync(taskRoot+'/comparison-final.json',JSON.stringify(report,null,2));console.log(JSON.stringify({exactOutputEquivalence:report.exactOutputEquivalence,mismatches,summaryBefore:before.summary,summaryAfter:after.summary,timings,detail},null,2));if(mismatches.length)process.exitCode=1;
