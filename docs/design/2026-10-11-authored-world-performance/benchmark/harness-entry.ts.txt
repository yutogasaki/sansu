function clean(value:any,seen=new Set<any>()):any {
    if(value===undefined)return '[undefined]';if(value===null||typeof value==='string'||typeof value==='boolean')return value;
    if(typeof value==='number')return Number.isFinite(value)?Object.is(value,-0)?0:value:String(value);
    if(typeof value==='function')return `[function:${value.name}]`;if(ArrayBuffer.isView(value))return typed(value);
    if(value instanceof T.Color)return {kind:'Color',value:value.toArray()};
    if(value.isVector2||value.isVector3||value.isVector4||value.isQuaternion||value.isMatrix3||value.isMatrix4)return {kind:value.constructor.name,value:value.toArray().map((n:number)=>clean(n))};
    if(value.isTexture)return texture(value);

}
