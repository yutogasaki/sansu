"""Preserve attempt02 raw failure, successful acquisitions, and preparation without rerunning."""
from pathlib import Path
from datetime import datetime, timezone
import json, hashlib, re

SOURCE = Path(__file__).resolve().parent
DEST = Path('/Users/yutogasaki/Projects/sansu/docs/design/2026-10-12-authored-shoulder-runtime/c2/floor-diagnostic/roi-attempt-02')
sha = lambda b: hashlib.sha256(b).hexdigest()
files = sorted(p for p in SOURCE.rglob('*') if p.is_file())
before = {str(p.relative_to(SOURCE)): sha(p.read_bytes()) for p in files}
assert before['actual-02/report.json'] == '09e097da6d2a0fed579ba8604baa147af654b46e0c0a6dd7ddb07f899366ed05'
assert not DEST.exists(), 'Do not overwrite earlier raw evidence'
DEST.mkdir(parents=True)
records = []
for source in files:
    relative = source.relative_to(SOURCE)
    archived = Path(str(relative)+'.txt') if source.suffix in ['.py', '.mjs'] else relative
    target = DEST/'raw'/archived
    target.parent.mkdir(parents=True, exist_ok=True)
    body = source.read_bytes()
    with target.open('xb') as f: f.write(body)
    assert sha(target.read_bytes()) == before[str(relative)]
    records.append({'sourceRelativePath': str(relative), 'archiveRelativePath': str(Path('raw')/archived), 'bytes': len(body), 'sha256': sha(body)})
after = {str(p.relative_to(SOURCE)): sha(p.read_bytes()) for p in files}
assert before == after
proof = {'recordedAtUtc': datetime.now(timezone.utc).isoformat(), 'source': str(SOURCE), 'archive': str(DEST),
         'fileCount': len(records), 'bytes': sum(r['bytes'] for r in records), 'files': records,
         'sourceBefore': before, 'sourceAfter': after, 'sourceUnchanged': before == after, 'allArchivedBytesMatch': True,
         'rawFailurePreserved': True, 'programFileSuffixOnly': '.mjs/.py -> .mjs.txt/.py.txt'}
with (DEST/'archive-proof.json').open('x') as f:f.write(json.dumps(proof,indent=2)+'\n')
