"""One bounded local-normal diagnostic; no execution without --start."""
import argparse, hashlib, json, os, signal, subprocess, time
from datetime import datetime, timezone
from pathlib import Path
HERE = Path(__file__).resolve().parent
EXPECTED = {'diagnose-roi-face-normal.mjs': 'f94d8ce0d306fc5c04228cfb63dcd81c47fdd7a533b697edc98fe24ec6fc3415', 'target.json': 'd7ecf37dfdce5eec6b7d331a1faf6afa446a34c9ef01e5581ea9f4a0f0f9d819'}
p = argparse.ArgumentParser()
p.add_argument('--start', action='store_true')
a = p.parse_args()
DRIVER = HERE / 'diagnose-roi-face-normal.mjs'
RUN = HERE / 'execution-01'
OUT = HERE / 'actual-01'
cmd = ['node', str(DRIVER), '--start', '--output', str(OUT)]
plan = {'command':cmd, 'seconds':900, 'processTreeRssBytes':4*1024**3, 'rssSampleSeconds':.5,
        'expectedCaptures':['shadow-off','roi-face-normal','restored-shadow-off'], 'sourceCount':6460,
        'installedDependencyCount':6, 'freshOutput':str(OUT), 'server':'Root-owned 5359, never signalled',
        'scope':'Causal diagnosis only; no product performance, art acceptance or source mutation'}
if not a.start:
 print(json.dumps({'prepared':True,'executionStarted':False,**plan},indent=2))
 raise SystemExit(0)
for name,value in EXPECTED.items(): assert hashlib.sha256((HERE/name).read_bytes()).hexdigest()==value,name
assert not RUN.exists() and not OUT.exists(), 'Preserve first attempt; no automatic retry'
RUN.mkdir()
(RUN/'plan.json').write_text(json.dumps(plan,indent=2)+'\n')
(RUN/'executed-wrapper.py.txt').write_bytes(Path(__file__).read_bytes())
started = datetime.now(timezone.utc).isoformat(); start = time.monotonic(); peak = 0; stop = None
samples=[]; cleanup=[]
with (RUN/'execution.log.txt').open('xb') as log:
 proc = subprocess.Popen(cmd,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
 pid=proc.pid
 try:
  while proc.poll() is None:
   rows=subprocess.run(['ps','-axo','pid=,ppid=,pgid=,rss='],capture_output=True,text=True,check=True).stdout.splitlines()
   table=[tuple(map(int,row.split())) for row in rows if len(row.split())==4]
   ids={pid}
   for _ in range(12): ids.update(child for child,parent,group,rss in table if parent in ids)
   owned=[row for row in table if row[0] in ids or row[2]==pid]
   rss=sum(row[3] for row in owned)*1024;peak=max(peak,rss)
   samples.append({'elapsedSeconds':time.monotonic()-start,'rssBytes':rss,'processCount':len(owned)})
   if rss>plan['processTreeRssBytes']:stop='Observed process-tree RSS 4 GiB boundary'
   if time.monotonic()-start>900:stop='900 second wall boundary'
   if stop:break
   time.sleep(.5)
 finally:
  if proc.poll() is None:
   cleanup.append({'signal':'SIGTERM','group':pid});os.killpg(pid,signal.SIGTERM)
   try:proc.wait(timeout=10)
   except subprocess.TimeoutExpired:
    cleanup.append({'signal':'SIGKILL','group':pid});os.killpg(pid,signal.SIGKILL);proc.wait()
code=proc.wait()
# Do not touch the separate root-owned Vite group. Only record any surviving owned group.
rows=subprocess.run(['ps','-axo','pid=,ppid=,pgid=,rss='],capture_output=True,text=True,check=True).stdout.splitlines()
remaining=[tuple(map(int,row.split())) for row in rows if len(row.split())==4 and int(row.split()[2])==pid]
unexpected_remaining=list(remaining)
if remaining:
 cleanup.append({'signal':'SIGTERM','group':pid,'reason':'Owned group survived driver exit'});os.killpg(pid,signal.SIGTERM)
 until=time.monotonic()+5
 while remaining and time.monotonic()<until:
  time.sleep(.5)
  rows=subprocess.run(['ps','-axo','pid=,ppid=,pgid=,rss='],capture_output=True,text=True,check=True).stdout.splitlines()
  remaining=[tuple(map(int,row.split())) for row in rows if len(row.split())==4 and int(row.split()[2])==pid]
 if remaining:
  cleanup.append({'signal':'SIGKILL','group':pid,'reason':'Owned group survived TERM grace'});os.killpg(pid,signal.SIGKILL)
  time.sleep(.5)
  rows=subprocess.run(['ps','-axo','pid=,ppid=,pgid=,rss='],capture_output=True,text=True,check=True).stdout.splitlines()
  remaining=[tuple(map(int,row.split())) for row in rows if len(row.split())==4 and int(row.split()[2])==pid]
after={name:hashlib.sha256((HERE/name).read_bytes()).hexdigest() for name in EXPECTED}
raw_path=OUT/'report.json';raw=json.loads(raw_path.read_text()) if raw_path.exists() else None
contract=bool(raw and raw.get('diagnosticCompleted') and raw.get('restoration',{}).get('complete')
 and raw.get('sourceFixed') and raw.get('installedFixed') and raw.get('driverDependenciesFixed')
 and [c['stage'] for c in raw.get('captures',[])]==plan['expectedCaptures'])
record={'command':cmd,'startedAtUtc':started,'finishedAtUtc':datetime.now(timezone.utc).isoformat(),
 'elapsedSeconds':time.monotonic()-start,'ownedPid':pid,'ownedProcessGroup':pid,'exitCode':code,
 'stopReason':stop,'peakObservedProcessTreeRssBytes':peak,'limits':plan,'inputBefore':EXPECTED,'inputAfter':after,
 'cleanupSignals':cleanup,'unexpectedOwnedGroupAfterDriverExit':unexpected_remaining,'ownedGroupRemaining':remaining,'rawReportPresent':raw_path.exists(),'rawContractPass':contract,
 'rawReportSha256':hashlib.sha256(raw_path.read_bytes()).hexdigest() if raw_path.exists() else None,
 'pass':code==0 and not stop and after==EXPECTED and contract and not unexpected_remaining and not remaining,
 'scope':'Diagnostic completion only; original screenshots require root visual judgment; not art/M1 acceptance'}
(RUN/'rss-samples.json').write_text(json.dumps(samples,indent=2)+'\n')
(RUN/'result.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record),flush=True)
raise SystemExit(0 if record['pass'] else 1)
