"""Scheduled one-process execution only. Fresh log/output; no automatic retries."""
import argparse,json,os,signal,subprocess,sys,time
from pathlib import Path
p=argparse.ArgumentParser();p.add_argument('--scheduled',action='store_true');p.add_argument('--seconds',type=int,required=True);p.add_argument('--rss-mib',type=int,default=2560);p.add_argument('--log',type=Path,required=True);p.add_argument('command',nargs=argparse.REMAINDER);a=p.parse_args()
assert a.scheduled and a.command;cmd=a.command[1:] if a.command[0]=='--' else a.command;assert cmd
assert not a.log.exists() and not Path(str(a.log)+'.json').exists(),'Preserve first attempt; select a fresh log'
a.log.parent.mkdir(parents=True,exist_ok=True);start=time.monotonic();peak=0;stop=None
with a.log.open('xb') as log:
 proc=subprocess.Popen(cmd,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
 while proc.poll() is None:
  time.sleep(1)
  rows=subprocess.check_output(['ps','-axo','pid=,ppid=,rss='],text=True).splitlines();table=[]
  for row in rows:
   try:table.append(tuple(map(int,row.split())))
   except ValueError:pass
  ids={proc.pid}
  for _ in range(5):ids.update(pid for pid,parent,rss in table if parent in ids)
  rss=sum(rss for pid,parent,rss in table if pid in ids)*1024;peak=max(peak,rss)
  if rss>a.rss_mib*1024*1024:stop='RSS reserve boundary'
  if time.monotonic()-start>a.seconds:stop='Wall-time boundary'
  if stop:
   os.killpg(proc.pid,signal.SIGTERM)
   try:proc.wait(timeout=5)
   except subprocess.TimeoutExpired:os.killpg(proc.pid,signal.SIGKILL);proc.wait()
   break
report={'command':cmd,'exitCode':proc.wait(),'stop':stop,'peakSampledRssBytes':peak,'elapsedSeconds':time.monotonic()-start,'limits':{'seconds':a.seconds,'rssMiB':a.rss_mib},'sourceOnly':True}
Path(str(a.log)+'.json').write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(report));sys.exit(0 if report['exitCode']==0 and not stop else 1)
