"""Tiny synthetic arithmetic checks. No source geometry, bpy or world compilation."""
from collections import Counter
import json
from entry_geometry import Terrain,ribbon_plan,solid_geometry,slope_report

def quad(x0,x1,z0,z1,height):
 p=[(x0,height(x0,z0),z0),(x1,height(x1,z0),z0),(x1,height(x1,z1),z1),(x0,height(x0,z1),z1)]
 return [(p[0],p[1],p[2]),(p[0],p[2],p[3])]
checks=[]
terrain=Terrain(quad(-2,2,-2,2,lambda x,z:.1*x+.05*z))
plan=ribbon_plan([(0,0,0),(0,.2,1)],.64,terrain)
assert plan['ready'] and len(plan['entrySection']['segments'])>=1
lo,hi=plan['entrySection']['flatLandingGapRange'];assert abs(lo+.032)<1e-12 and abs(hi-.032)<1e-12
checks.append('sloping ground exposes both burial and floating ends of former flat landing')
actual=terrain.relation([[plan['topVertices'][i] for i in f] for f in plan['topFaces']]);assert actual['clear'] and abs(actual['minGap'])<1e-12
checks.append('conformed exact-width seam joins existing sloped plane, rising ribbon stays above it')
vertices,faces=solid_geometry(plan,.07);edges=Counter(tuple(sorted((a,b))) for f in faces for a,b in zip(f,f[1:]+f[:1]));assert all(n==2 for n in edges.values())
volume=sum((vertices[a][0]*(vertices[b][1]*vertices[c][2]-vertices[b][2]*vertices[c][1])+vertices[a][1]*(vertices[b][2]*vertices[c][0]-vertices[b][0]*vertices[c][2])+vertices[a][2]*(vertices[b][0]*vertices[c][1]-vertices[b][1]*vertices[c][0]))/6 for a,b,c in faces)
assert volume>0
checks.append('planned ribbon closes each edge exactly twice with outward winding')
flat=[list(v) for v in plan['topVertices']]
for v in flat[:len(plan['columns'])]:v[1]=0
assert not terrain.relation([[flat[i] for i in f] for f in plan['topFaces']])['clear']
checks.append('former horizontal seam is rejected when it buries part of its width')
hole=Terrain(quad(-2,-.1,-2,2,lambda x,z:0)+quad(.1,2,-2,2,lambda x,z:0))
assert not ribbon_plan([(0,0,0),(0,.2,1)],.64,hole)['ready']
checks.append('centre/width gap in original terrain cannot synthesize a landing')
step=Terrain(quad(-2,0,-2,2,lambda x,z:0)+quad(0,2,-2,2,lambda x,z:.03))
assert not ribbon_plan([(0,.03,0),(0,.2,1)],.64,step)['ready']
checks.append('discontinuous existing height seam is HOLD, not smoothed into invented ground')
assert not terrain.relation(terrain.triangles)['clear']
checks.append('positive-area duplicated source top is rejected')
assert slope_report([(0,0,0),(0,.2,.1)])['maximumAbsoluteSlope']==2
checks.append('short entry connector grade is measured rather than hidden by fixed arc grade')
print(json.dumps({'scope':'synthetic static arithmetic only','checks':checks,'passed':len(checks),'sourceGeometryMeasured':False,'BlenderExecuted':False,'worldCompiled':False},indent=2))
