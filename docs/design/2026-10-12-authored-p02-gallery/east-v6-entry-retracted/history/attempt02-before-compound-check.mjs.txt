    assert(w && size && !a.sparse);
    return Array.from({ length: a.count }, (_, i) => Array.from({ length: w }, (_, k) => {
        const off = (v.byteOffset ?? 0) + (a.byteOffset ?? 0) + i * (v.byteStride ?? size * w) + k * size;
        return a.componentType === 5126 ? sourceBin.readFloatLE(off) : a.componentType === 5123 ? sourceBin.readUInt16LE(off) : sourceBin.readUInt32LE(off);
    }));
}

function topology(rows) {
    const edges = new Map(); let zero = 0, volume = 0;
    for (const { positions: t } of rows) {
        const n = cross(...t); if (!n.lengthSq()) zero++;
        volume += new T.Vector3(...t[0]).dot(new T.Vector3(...t[1]).cross(new T.Vector3(...t[2]))) / 6;
        for (let i = 0; i < 3; i++) {
            const a = key(t[i]), b = key(t[(i + 1) % 3]), k = [a, b].sort().join('|');
            const r = edges.get(k) ?? { count: 0, balance: 0, a: t[i], b: t[(i + 1) % 3] };
            r.count++; r.balance += a < b ? 1 : -1; edges.set(k, r);
        }
    }
    return { triangles: rows.length, zeroArea: zero, signedVolume: volume,
        badEdges: [...edges.values()].filter(e => e.count !== 2 || e.balance !== 0).length,
        boundary: [...edges.values()].filter(e => e.count === 1) };
}
function solid(topTriangles) {
    const top = topTriangles.map(t => flat(upward(t))), body = [];
    for (const r of top) body.push(flat(r.positions.map(p => [p[0], Math.fround(p[1] - (H - BOTTOM)), p[2]]).reverse()));
    for (const e of topology(top).boundary) {
        const a = e.a, b = e.b, c = [a[0], Math.fround(a[1] - (H - BOTTOM)), a[2]], d = [b[0], Math.fround(b[1] - (H - BOTTOM)), b[2]];
        body.push(flat([a, c, b]), flat([b, c, d]));
    }
    return { top, body };
}
