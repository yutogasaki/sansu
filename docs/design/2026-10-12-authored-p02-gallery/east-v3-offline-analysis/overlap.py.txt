from pathlib import Path
import json,hashlib,collections
from shapely.geometry import Polygon
from shapely.strtree import STRtree
P=Path(__file__).resolve().parent;V=P.parent/'east-v3-manifold';raw=V/'runs/01-coarse-build/planned-floor-tops.json';d=json.loads(raw.read_text());result={}
for name,p in d.items():
 triangles=[[p['topVertices'][i] for i in f] for f in p['topFaces']];polys=[Polygon([(v[0],v[2]) for v in t]) for t in triangles];tree=STRtree(polys);pairs=[]
 for i,a in enumerate(polys):
  for j in tree.query(a):
   j=int(j)
   if j<=i:continue
   part=a.intersection(polys[j]);area=part.area
