"""Tiny static/synthetic checks; never imports Blender or runs source geometry."""
from pathlib import Path
import ast,json,hashlib,sys,collections,math
P=Path(__file__).resolve().parent;BASE=P.parent;sys.path.insert(0,str(P))
from entry_geometry import Terrain,EPS
from terrain_connector import terrain_connector,closed_solid
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
S=json.loads((P/'study.json').read_text());V=json.loads((BASE/'east-v1/study.json').read_text());M=json.loads((BASE/'east-v1/manifest.json').read_text())
checks={}
for f in P.glob('*.py'):ast.parse(f.read_text())
checks['pythonAST']=True
checks['sealedV1FilesUnchanged']=all(sha(BASE/'east-v1'/name)==v['sha256'] for name,v in M['files'].items())
checks['eightSourceInputsUnchanged']=all(sha(v['path'])==v['sha256'] for v in S['inputs'].values())
checks['reviewedMeasurementAndProbeUnchanged']=all(sha(v['path'])==v['sha256'] for v in S['basis'].values())
checks['ownerTransformEnvelopeFixed']=all(S[k]==V[k] for k in ['transform','envelope','place','inputs','scope','camera'])
checks['widthHeadEpsilonNotRelaxed']=all(S['design'][k]==V['design'][k] for k in ['clearWidth','structuralWidth','headReserve','probeStep','thickness']) and sha(P/'entry_geometry.py')==sha(BASE/'east-v1/entry_geometry.py')
a=(P/'build-coarse.py').read_text();b=(BASE/'east-v1/source-study.py').read_text();start="    donor_names=(D['sourceFloorName']";end='    interior,old_gallery,bark_donor,original_trunk_probe=donors'
checks['donorTransformBlockByteIdentical']=a[a.index(start):a.index(end)]==b[b.index(start):b.index(end)]
checks['restoredLeftObstaclesRemain']=next(x for x in a.splitlines() if 'obstacles=[' in x)==next(x for x in b.splitlines() if 'obstacles=[' in x)
checks['boughEndDoesNotPierceWalkTop']=S['design']['boughRadius']<S['design']['thickness'] and "dest=p-Vector((0,D['thickness'],0))" in a
checks['noRenderInBuild']='bpy.ops.render' not in a
checks['statusAlwaysHold']="'status':'HOLD'" in a and 'HOLD_COARSE_SOURCE_FOR_WHOLE_AND_ENTRY_REVIEW' in a
# Synthetic tilted ground, two source triangles. Full threshold matches original
# terrain; fixed high end stays above it. No actual source build occurs here.
verts=[[-2,-.2,-2],[2,.2,-2],[2,.2,2],[-2,-.2,2]];terrain=Terrain([[verts[i] for i in f] for f in [[0,2,1],[0,3,2]]])
plan=terrain_connector(terrain,[0,0,0],[0,.5,1],.64,.5);v,f,qa=closed_solid(plan,.07)
edge=collections.Counter(tuple(sorted((a,b))) for face in f for a,b in zip(face,face[1:]+face[:1]))
checks['syntheticClosedFloor']=all(n==2 for n in edge.values())
checks['syntheticNoBuriedTop']=plan['topMinusGroundRange'][0]>=-EPS
checks['syntheticActualThresholdRange']=abs(plan['entrySection']['flatLandingGapRange'][0]+.032)<EPS and abs(plan['entrySection']['flatLandingGapRange'][1]-.032)<EPS
checks['syntheticRejectsTargetBelowGround']=False
try:terrain_connector(terrain,[0,0,0],[0,-.5,1],.64,-.5)
except AssertionError:checks['syntheticRejectsTargetBelowGround']=True
# Original cap is terminal at95deg; the new visible floor115deg adds real
# forward support room while leaving the owner/use target unchanged.
r=S['design']['shortGalleryCenterRadius'];reserve=S['design']['clearWidth']/2
checks['terminalRailPlaneHasNominalReserve']=r*math.sin(math.radians(115-95))-S['design']['railRadius']>reserve
assert all(checks.values()),checks
print(json.dumps({'status':'STATIC_SYNTHETIC_ONLY_PASS','checks':checks,'sourceBuildExecuted':False,'syntheticVertices':len(v),'syntheticFaces':len(f),'pending':'Blender boolean topology, exact seams after Float32, source guards, support and views'},indent=2))
