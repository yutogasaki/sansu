"""One explicit visible timber threshold on fixed terrain. No bpy/app imports."""
import math,struct
from entry_geometry import clip,polygon_area,plane_y,EPS

def terrain_connector(terrain,entry,end,width,top_y):
    dx,dz=end[0]-entry[0],end[2]-entry[2];run=math.hypot(dx,dz);assert run>0
    tx,tz=dx/run,dz/run;sx,sz=-tz,tx
    a=[entry[0]-sx*width/2,entry[1],entry[2]-sz*width/2];b=[entry[0]+sx*width/2,entry[1],entry[2]+sz*width/2]
    c=[end[0]+sx*width/2,top_y,end[2]+sz*width/2];d=[end[0]-sx*width/2,top_y,end[2]-sz*width/2]
    quad=[a,b,c,d];vertices=[];faces=[];lineage=[];covered=0;gaps=[];section=terrain.section(a,b,entry[1])
    assert section['supportedFullWidth'],'Ground threshold has uncovered/discontinuous terrain'
    for index in terrain.nearby(quad):
        ground=terrain.triangles[index];poly=clip(quad,ground)
        area=polygon_area(poly) if len(poly)>=3 else 0
        if area<=1e-12:continue
        covered+=area;base=len(vertices)
        for point in poly:
            y0=plane_y(point,ground);u=max(0,min(1,((point[0]-entry[0])*tx+(point[2]-entry[2])*tz)/run));y=y0+(top_y-y0)*u
            gaps.append(y-y0);assert y>=y0-EPS,'Fixed native interior is below terrain in this candidate; do not auto-raise floor'
            vertices.append([point[0],y,point[2]])
        for i in range(1,len(poly)-1):
            face=[base,base+i,base+i+1]
            pa,pb,pc=[vertices[j] for j in face]
            crossxz=(pb[0]-pa[0])*(pc[2]-pa[2])-(pb[2]-pa[2])*(pc[0]-pa[0])
            if crossxz>0:face.reverse()
            assert abs(crossxz)>1e-14,'Degenerate terrain-clipped top; keep failure'
            faces.append(face);lineage.append(index)
    assert abs(covered-run*width)<=EPS,'Single terrain-domain area differs; no invented floor filling'
    return {'topVertices':vertices,'topFaces':faces,'sourceGroundTrianglePerFace':lineage,'entrySection':section,'coveredArea':covered,'domainArea':run*width,'topMinusGroundRange':[min(gaps),max(gaps)],'construction':'clip one chosen corridor at every real terrain triangle; lift linearly from exact threshold to fixed native interior, no terrain edits'}

def closed_solid(plan,thickness):
    # Clipping adjacent source triangles can produce 1e-16 differences at
    # one shared intersection. Reuse a vertex only within the existing EPS;
    # keep its first coordinate, record the distance, never move source ground.
    vertices=[];raw_vertices=[];buckets={};float_keys={};remap=[];faces=[];max_weld=0.
    for p in plan['topVertices']:
        cell=tuple(math.floor(x/EPS) for x in p);found=None
        for dx in [-1,0,1]:
            for dy in [-1,0,1]:
                for dz in [-1,0,1]:
                    for index in buckets.get((cell[0]+dx,cell[1]+dy,cell[2]+dz),[]):
                        distance=math.dist(raw_vertices[index],p)
                        if distance<=EPS:found=index;max_weld=max(max_weld,distance);break
        if found is None:
            key=struct.pack('<3f',*p)
            assert key not in float_keys,'Float32 coalesces distinct intersections beyond existing EPS; retain HOLD'
            found=len(vertices);vertices.append(list(struct.unpack('<3f',key)));raw_vertices.append(p);buckets.setdefault(cell,[]).append(found);float_keys[key]=found
        remap.append(found)
    for face in plan['topFaces']:
        q=[remap[i] for i in face];assert len(set(q))==3,'Float32 collapsed new top triangle; source must remain HOLD';faces.append(q)
    edges={}
    for face in faces:
        for a,b in zip(face,face[1:]+face[:1]):edges.setdefault(tuple(sorted((a,b))),[]).append((a,b))
    assert all(len(x)<=2 for x in edges.values()),'Overlapping/nonmanifold terrain top'
    boundary=[x[0] for x in edges.values() if len(x)==1]
    count=len(vertices);vertices+=[[x,y-thickness,z] for x,y,z in vertices]
    result=list(faces)+[[c+count,b+count,a+count] for a,b,c in faces]
    for a,b in boundary:result.extend([[b,a,a+count],[b,a+count,b+count]])
    return vertices,result,{'topFaces':len(faces),'boundaryEdges':len(boundary),'float32UniqueTopVertices':count,'rawTopVertices':len(plan['topVertices']),'sourceTopTrianglesRetained':len(faces)==len(plan['topFaces']),'intersectionWeldEpsilon':EPS,'maximumIntersectionWeldDistance':max_weld}
