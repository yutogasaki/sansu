def polygon_at_band(tri,ymin,ymax):
 poly=tri
 for y,lower in [(ymin,True),(ymax,False)]:
  out=[]
  for a,b in zip(poly,poly[1:]+poly[:1]):
   ia=a[1]>=y if lower else a[1]<=y;ib=b[1]>=y if lower else b[1]<=y
   if ia:out.append(a)
   if ia!=ib:
    u=(y-a[1])/(b[1]-a[1]);out.append([a[k]+(b[k]-a[k])*u for k in range(3)])
  poly=out
 return poly
def projected_distance(poly,p):
 if not poly:return None
 values=[];positive=negative=False;area=0
 for a,b in zip(poly,poly[1:]+poly[:1]):
  dx,dz=b[0]-a[0],b[2]-a[2];cr=dx*(p[2]-a[2])-dz*(p[0]-a[0]);positive|=cr>1e-8;negative|=cr<-1e-8;den=dx*dx+dz*dz;t=max(0,min(1,((p[0]-a[0])*dx+(p[2]-a[2])*dz)/den))if den else 0;v=[a[0]+t*dx,a[2]+t*dz];values.append((math.hypot(v[0]-p[0],v[1]-p[2]),v));area+=a[0]*b[2]-b[0]*a[2]
 if abs(area)>1e-8 and not(positive and negative):return {'distance':0.,'xz':[p[0],p[2]],'insideProjectedPolygon':True}
 d,v=min(values);return {'distance':d,'xz':v,'insideProjectedPolygon':False}
