def mesh_identity():
    out={}
    for obj in bpy.data.objects:
        if obj.type!='MESH':continue
        h=hashlib.sha256()
        for row in obj.matrix_world:h.update(struct.pack('<4d',*row))
        for v in obj.data.vertices:h.update(struct.pack('<3f',*v.co))
        for polygon in obj.data.polygons:
            h.update(struct.pack('<I',len(polygon.vertices)))
            for index in polygon.vertices:h.update(struct.pack('<I',index))
        h.update(json.dumps({'hideRender':obj.hide_render,'hideViewport':obj.hide_viewport,'materials':[m.name if m else None for m in obj.data.materials]},sort_keys=True).encode());out[obj.name]=h.hexdigest()
    return out
def material_identity():
    def scalar(value):
        if isinstance(value,(str,int,float,bool)) or value is None:return value
        try:return [scalar(x) for x in value]
        except TypeError:return str(value)
    result={}
    for material in bpy.data.materials:
        nodes=[];links=[]
        if material.use_nodes:
            for node in material.node_tree.nodes:
                nodes.append({'name':node.name,'type':node.bl_idname,'inputs':{str(i)+':'+s.name:scalar(s.default_value) for i,s in enumerate(node.inputs) if hasattr(s,'default_value')}})

        result[material.name]={'diffuseColor':list(material.diffuse_color),'useNodes':material.use_nodes,'nodes':nodes,'links':links}
    return result
