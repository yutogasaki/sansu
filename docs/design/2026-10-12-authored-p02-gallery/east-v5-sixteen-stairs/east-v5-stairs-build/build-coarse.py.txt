def digest(obj):
    h = hashlib.sha256()
    for row in obj.matrix_world: h.update(struct.pack('<4d', *row))
    if obj.type == 'MESH':
        for v in obj.data.vertices: h.update(struct.pack('<3f', *v.co))
        for loop in obj.data.loops: h.update(struct.pack('<I', loop.vertex_index))
        for n in obj.data.corner_normals: h.update(struct.pack('<3f', *n.vector))
        for m in obj.data.materials: h.update((m.name if m else '').encode())
    return h.hexdigest()

def vertices_faces(obj):
    deps = bpy.context.evaluated_depsgraph_get(); ev = obj.evaluated_get(deps); mesh = ev.to_mesh()
    try:
        mesh.calc_loop_triangles(); matrix = BI @ ev.matrix_world
        return [matrix @ v.co for v in mesh.vertices], [tuple(t.vertices) for t in mesh.loop_triangles]
    finally: ev.to_mesh_clear()

def matrix_rows(matrix): return [list(row) for row in matrix]

def donor_snapshot(obj, reference, expected_matrix_gltf):
    """Pair an evaluated mesh with its own evaluated matrix, never a stale pose."""
    deps=bpy.context.evaluated_depsgraph_get();ev=obj.evaluated_get(deps);mesh=ev.to_mesh()
    source_matrix=ev.matrix_world.copy()
    try:
        mesh.calc_loop_triangles()
        local=[list(vertex.co) for vertex in mesh.vertices]
        local_gltf=[transform_point(matrix_rows(BI),p) for p in local]
        actual_world=[transform_point(matrix_rows(BI @ source_matrix),p) for p in local]
        local_check=compare_bounds(bounds(local_gltf),reference['localBoundsGltf'],EPS)
        matrix_check=compare_matrix(matrix_rows(BI @ source_matrix @ B),expected_matrix_gltf,EPS)
        world_check=compare_bounds(bounds(actual_world),transform_bounds(expected_matrix_gltf,reference['localBoundsGltf']),EPS)
        h=hashlib.sha256()
        for point in local:
            h.update(struct.pack('<3f',*point))
        for triangle in mesh.loop_triangles:
            h.update(struct.pack('<3I',*triangle.vertices))
        record={'name':obj.name,'referenceName':reference['name'],'gltfNode':reference['gltfNode'],
                'parent':obj.parent.name if obj.parent else None,'constraints':[c.type for c in obj.constraints],
                'matrixBasis':matrix_rows(obj.matrix_basis),'matrixWorld':matrix_rows(obj.matrix_world),
                'matrixParentInverse':matrix_rows(obj.matrix_parent_inverse),'evaluatedMatrixWorld':matrix_rows(source_matrix),
                'evaluatedVertices':len(local),'evaluatedTriangles':len(mesh.loop_triangles),'localGeometrySha256':h.hexdigest(),
                'localBoundsCheck':local_check,'sourceOrAuthoredMatrixCheck':matrix_check,'worldBoundsCheck':world_check,
                'matches':not obj.parent and not obj.constraints and local_check['matches'] and matrix_check['matches'] and world_check['matches'],
                'arithmetic':'bounds evaluated in double precision from captured Float32 mesh/matrix; unchanged EPS 1e-7'}
        return record,source_matrix
    finally: ev.to_mesh_clear()

def tree(objects, local=False, provenance=False):
    vertices, faces, sources = [], [], []
    for obj in objects:
        if obj.type != 'MESH': continue
        if local:
            corners = [BI @ obj.matrix_world @ Vector(p) for p in obj.bound_box]
            if max(p.x for p in corners)<1.6 or min(p.x for p in corners)>5.4 or max(p.z for p in corners)<-10.8 or min(p.z for p in corners)>-6.8: continue
        vs, fs = vertices_faces(obj); base=len(vertices); vertices.extend(vs)
        for face in fs:
            if local:
                pts=[vs[i] for i in face]
                if max(p.x for p in pts)<1.6 or min(p.x for p in pts)>5.4 or max(p.z for p in pts)<-10.8 or min(p.z for p in pts)>-6.8: continue
            faces.append(tuple(base+i for i in face))
            if provenance:sources.append({'object':obj.name,'triangle':[list(vs[i]) for i in face]})
    assert faces, 'Missing real source geometry'
    result=(BVHTree.FromPolygons(vertices, faces, all_triangles=True), len(faces))
    return (*result,sources) if provenance else result

def ground_at(x,z):

def box_tree(center, radius, height):
    # Conservative full-width sample, not an exact moving-rig acceptance proof.
    x,y,z=center
    vs=[(x+dx*radius,y+dy,z+dz*radius) for dy in (.012,height) for dz in (-1,1) for dx in (-1,1)]
    fs=[(0,1,3,2),(4,6,7,5),(0,4,5,1),(2,3,7,6),(0,2,6,4),(1,5,7,3)]
    return BVHTree.FromPolygons(vs, fs)

def resample(points):
    out=[points[0].copy()]
    for a,b in zip(points,points[1:]):
        count=max(1,math.ceil((b-a).length/D['probeStep']))
        out.extend(a.lerp(b,i/count) for i in range(1,count+1))
    return out

def tube(name, points, radius):
    sides=8;v=[];f=[]
    for i,p in enumerate(points):
        tangent=(points[min(i+1,len(points)-1)]-points[max(i-1,0)]).normalized();side=tangent.cross(Vector((0,1,0)))
        if side.length<.001:side=tangent.cross(Vector((1,0,0)))
        side.normalize();across=tangent.cross(side).normalized()
        for j in range(sides):v.append(p+radius*(math.cos(j*math.tau/sides)*side+math.sin(j*math.tau/sides)*across))
    for i in range(len(points)-1):
        for j in range(sides):f.append((i*sides+j,i*sides+(j+1)%sides,(i+1)*sides+(j+1)%sides,(i+1)*sides+j))
    f += [tuple(reversed(range(sides))),tuple((len(points)-1)*sides+j for j in range(sides))]
    return new_mesh(name,v,f,bark,'structure')
