def accessor(index):
    a=doc['accessors'][index];v=doc['bufferViews'][a['bufferView']]
    assert 'sparse' not in a and not a.get('normalized',False)
    typ={'SCALAR':1,'VEC3':3}[a['type']];code={5121:'B',5123:'H',5125:'I',5126:'f'}[a['componentType']]
    fmt='<'+code*typ;size=struct.calcsize(fmt);offset=v.get('byteOffset',0)+a.get('byteOffset',0);stride=v.get('byteStride',size)
    rows=[struct.unpack_from(fmt,binary,offset+i*stride) for i in range(a['count'])]
    return rows,sha(b''.join(struct.pack(fmt,*r) for r in rows))

def exact_zero(positions,faces):
    # Float32 source values are exact dyadic rationals. Shared binary denominator
    # permits exact integer cross products without a geometric epsilon.
    ratios=[[float(x).as_integer_ratio() for x in p] for p in positions]
    den=max(q for p in ratios for _,q in p)
    points=[[n*(den//q) for n,q in p] for p in ratios]
    zero=[];areas=[]
    for fi,face in enumerate(faces):
        a,b,c=[points[i] for i in face];u=[y-x for x,y in zip(a,b)];v=[y-x for x,y in zip(a,c)]
        cross=[u[1]*v[2]-u[2]*v[1],u[2]*v[0]-u[0]*v[2],u[0]*v[1]-u[1]*v[0]]
        if cross==[0,0,0]:zero.append(fi)
        else:areas.append(math.sqrt(sum((x/(den*den))**2 for x in cross))/2)
    return zero,min(areas,default=None)
