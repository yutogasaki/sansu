import ast

import json

    ast.parse(p.read_text(),filename=str(p));syntax.append(p.name)

       BASE/'extraction-attempt02/fixed-c1-context.glb',BASE/'extraction-attempt02/bindings.json']

        'files':files,'externalInputs':external,'syntaxParsed':syntax,
        'command':'SANSU_FOREST_REPRESENTATION_BUILD_APPROVED=1 python3 '+str(HERE/'run-once.py'),
        'threads':2,'rssLimitBytes':3*1024**3,'wallSeconds':300,

        'firstFailureNeverPromotedToSuccess':True,'wholeView':'separate root decision after this build',
        'blenderExecuted':False}

digest=sha(HERE/'readiness.json');(HERE/'readiness.sha256').write_text(digest+'\n')
print(json.dumps({'readinessSha256':digest,'localFiles':len(files),'externalInputs':len(external),'syntaxFiles':len(syntax)}))
