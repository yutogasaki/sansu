def save(name,data):
    path=OUT/name;path.parent.mkdir(parents=True,exist_ok=True)
    raw=(json.dumps(data,indent=2,allow_nan=False)+'\n').encode()
    path.write_bytes(gzip.compress(raw,mtime=0) if path.suffix=='.gz' else raw)


def snapshot(objects,materials):
    return {'objects':{o.name:{'world':[list(r) for r in o.matrix_world],
        'basis':[list(r) for r in o.matrix_basis],'parentInverse':[list(r) for r in o.matrix_parent_inverse],
        'parent':o.parent.name if o.parent else None,'mesh':native.fixed_digest(o) if o.type=='MESH' else None,
        'sharp':edge_fingerprint((tuple(e.vertices),e.use_edge_sharp) for e in o.data.edges) if o.type=='MESH' else None}
        for o in objects},'materials':{m.name:checks.material_digest(m) for m in materials}}


def original_parity(label):
    after=snapshot(IMPORTED,MATERIALS);equal=after==source_before
    save(label,{'equal':equal,'guardedMeshes':sum(o.type=='MESH' for o in IMPORTED),
        'changedObjects':[n for n,v in source_before['objects'].items() if after['objects'][n]!=v],
        'changedMaterials':[n for n,v in source_before['materials'].items() if after['materials'][n]!=v],
        'repairOrRestore':False})
    assert equal,'Imported source geometry/parent/material changed'


def emit_role(key):
    def emit(name,data):
        save('roles/'+key+'/'+(name if name.endswith('.gz') else name+'.gz'),data)
        # Observe every clone before its next hard assertion, including a clone
        # in a role that subsequently fails NORMAL or another independent gate.
        if name.endswith('-evaluation.json') and 'precisionClassification' in data:
            row=data['precisionClassification']
            WORLD_EVALUATIONS.append(row);save('world-evaluations.json',WORLD_EVALUATIONS)
            if row['precisionHold']:
                PRECISION_HOLDS.append(row);save('precision-holds.json',PRECISION_HOLDS)
    return emit


def triangles(obj,normals):
    def yup(v):return [v[0],v[2],-v[1]]
    assert len(normals)==len(obj.data.loops)
    assert all(len(p.vertices)==3 for p in obj.data.polygons)
    return [{'positions':[yup(obj.data.vertices[obj.data.loops[i].vertex_index].co) for i in p.loop_indices],
             'normals':[yup(normals[i]) for i in p.loop_indices]} for p in obj.data.polygons]


def identity(obj):return str(obj['ownerId'])+':'+str(obj['sourceNodeIndex'])
def volume(obj):
    if obj.name not in VOLUMES:VOLUMES[obj.name]=Volume64(obj,BI)
    return VOLUMES[obj.name]
def role(owner,kind,ordinal=0):
    found=[o for o in ROLES if o['ownerId']==owner and o['forestRole']==kind and o['forestOrdinal']==ordinal]
    assert len(found)==1;return found[0]
def mapped(owner,kind,ordinal,point):return Vector(fields.field_for_role(owner,kind,ordinal,PROPOSAL)(point))


def connect(left_obj,right_obj,left_guess,right_guess,radius,max_span,style=None,donor_obj=None):
    # Unchanged stage4 primary-junction conditions and same19 maximum bough cap.
    assert left_obj['ownerId']==right_obj['ownerId']
    left,right=volume(left_obj),volume(right_obj)
    row={'left':left_obj.name,'right':right_obj.name,'ownerId':left_obj['ownerId'],
         'leftGuess':list(left_guess),'rightGuess':list(right_guess),'ballRadiusWorld':radius*.6,
         'status':'unresolved','clearanceApproximate':True};JOINTS.append(row)
    try:
        overlap=joints.overlap_ball(left,right,[left_guess,right_guess],radius*.6)
        if overlap:row.update(status='measured-overlap-under-inherited-tolerances',overlap=overlap)
        else:
            assert style in ('willow-main','willow-stem'),'Birch source contact unresolved; repair forbidden'
            assert len(CREATED)<PROPOSAL['guards']['maximumDerivedBoughs']
            assert sum(o.get('boughStyle')==style for o in CREATED)<PROPOSAL['guards']['maximumWillowMainBoughs' if style=='willow-main' else 'maximumWillowStemBoughs']
            assert donor_obj is not None
            obj,measurement=junctions.connector(left,right,left_guess,right_guess,radius,max_span,
                f"stage4-v2-junction-{len(JOINTS):02d}-{left_obj['ownerId']}",left_obj['ownerId'],donor_obj.data.materials[0],
                JOINT_COLLECTION,B,STUDY['sourceSha256'],volume(donor_obj),style,PROPOSAL['owners']['l14']['derivedBoughControls'])
            obj['boughStyle']=style;CREATED.append(obj)
            row.update(status='derived-closed-bough-approximate-endpoint-measurements',measurement=measurement)
    except BaseException as error:
        row['error']=repr(error)
        CREATED.extend(o for o in JOINT_COLLECTION.objects if o.type=='MESH' and o not in CREATED)
        raise
    finally:save('primary-junctions.json',JOINTS)
