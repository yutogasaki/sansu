def save(name,value):
    text=json.dumps(value,indent=2,allow_nan=False)+'\n'
    if name.endswith('.gz'):(OUT/name).write_bytes(gzip.compress(text.encode(),mtime=0))
    else:(OUT/name).write_text(text)

def vectors(values):return [list(v) for v in values]
def rows(matrix):return [list(row) for row in matrix]
def normal_stats(mesh,expected):
    actual=[n.vector.copy() for n in mesh.corner_normals]
    errors=[(a-b).length for a,b in zip(actual,expected)]
    assert len(actual)==len(expected)
    return {'maxL2':max(errors),'meanL2':sum(errors)/len(errors),'overOriginalGuard':sum(e>.002 for e in errors),
            'normalLengths':[min(n.length for n in actual),max(n.length for n in actual)]},actual

def clone(obj,data,name):
    result=obj.copy();result.data=data.copy();result.name=name
    bpy.context.scene.collection.objects.link(result)
    assert result.matrix_world==obj.matrix_world
    return result

def frame_locals(error):

    while tb:
        if Path(tb.tb_frame.f_code.co_filename).resolve()==(STAGE/'deform.py').resolve() and tb.tb_frame.f_code.co_name=='apply':
            return dict(tb.tb_frame.f_locals)
        tb=tb.tb_next
    raise AssertionError('Expected original deform.apply exception frame unavailable')
