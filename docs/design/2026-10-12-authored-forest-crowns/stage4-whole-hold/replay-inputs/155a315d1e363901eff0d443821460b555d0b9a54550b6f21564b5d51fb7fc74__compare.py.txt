def save(name, value):
    data = (json.dumps(value, indent=2, allow_nan=False) + '\n').encode()
    (OUT / name).write_bytes(gzip.compress(data, mtime=0) if name.endswith('.gz') else data)


def vecs(values):
    return [list(value) for value in values]


def errors(actual, expected):
    assert len(actual) == len(expected)
    return [(Vector(a) - Vector(b)).length for a, b in zip(actual, expected)]


def stats(actual, expected):
    values = errors(actual, expected)
    return {'maxL2': max(values), 'over002': sum(value > .002 for value in values), 'corners': len(values)}


def edge_key(edge):
    return ','.join(map(str, sorted(edge.vertices)))


def sharp(mesh):
    return {edge_key(edge): bool(edge.use_edge_sharp) for edge in mesh.edges}


def cross(a, b):
    return (a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0])


def sub(a, b):
    return tuple(x - y for x, y in zip(a, b))


def yup(value):
    # Blender local -> glTF local. Exact component/sign change, no rounding/weld.
    return [value[0], value[2], -value[1]]


def expected_triangles(mesh, normals):
    return [{'positions': [yup(mesh.vertices[mesh.loops[i].vertex_index].co) for i in polygon.loop_indices],
             'normals': [yup(normals[i]) for i in polygon.loop_indices]} for polygon in mesh.polygons]


def export_and_check(obj, desired, label):
    # Keep the original owner matrix/parent. Only this diagnostic clone selected.
    matrix_before = obj.matrix_world.copy()
    digest_before = native.fixed_digest(obj)
    desired_triangles = expected_triangles(obj.data, desired)
    readback_triangles = expected_triangles(obj.data, [item.vector.copy() for item in obj.data.corner_normals])
    path = OUT / (label + '-DIAGNOSTIC.glb')
    bpy.ops.object.select_all(action='DESELECT')
    obj.select_set(True)
    bpy.context.view_layer.objects.active = obj
    bpy.ops.export_scene.gltf(filepath=str(path), export_format='GLB', use_selection=True,
        export_normals=True, export_apply=False, export_yup=True, export_extras=True,
        export_animations=False, export_cameras=False, export_lights=False, export_materials='EXPORT')
    assert obj.matrix_world == matrix_before and native.fixed_digest(obj) == digest_before
    document, node, triangles, attributes = read_glb(path)
    readback = compare_triangles(triangles, readback_triangles)
    source = compare_triangles(triangles, desired_triangles)
    save(label + '-export-accessors.json.gz', {'node': node, 'allNodes': document['nodes'],
        'materials': document.get('materials', []), 'attributes': attributes,
        'expectedFromRawSource': desired_triangles, 'expectedFromBlenderReadback': readback_triangles})
    return {'file': path.name, 'sha256': sha(path), 'bytes': path.stat().st_size,
        'exportVsBlenderCornerReadback': readback, 'exportVsRawSourceDesired': source,
        'objectMatrixAndGeometryUnchangedByExport': True,
        'passed': source['physicalTriangleMultisetExact'] and source['normalMaxL2'] <= .002,
        'scope': 'local accessor triangle/corner parity; delivered instance/world composition not accepted'}


try:
    bpy.ops.wm.read_factory_settings(use_empty=True)

    original_digests = {obj.name: native.fixed_digest(obj) for obj in originals}
    original_flags = {obj.name: {'sharp': sharp(obj.data), 'smooth': [p.use_smooth for p in obj.data.polygons]} for obj in originals}

    faces = [tuple(polygon.vertices) for polygon in mesh.polygons]
    edges = [tuple(edge.vertices) for edge in mesh.edges]

    assert vecs(positions) == PRIOR['inputLocalPositions']

    zero = [i for i, face in enumerate(faces) if cross(sub(positions[face[1]], positions[face[0]]), sub(positions[face[2]], positions[face[0]])) == (0., 0., 0.)]
    assert zero == [0, 2, 98, 100], 'Exact-zero set changed; no near-zero removal'

    source_loops = [loop for i in keep for loop in mesh.polygons[i].loop_indices]
    provided = [source_normals[mesh.loops[i].vertex_index] for i in source_loops]
    original_sharp = sharp(mesh)

    def clone(name, requested_sharp):
        obj = original.copy()
        obj.data = bpy.data.meshes.new(name + '-mesh')
        obj.name = name
        obj.data.from_pydata(vecs(positions), edges, [faces[i] for i in keep])
        obj.data.update()
        for mat in mesh.materials:
            obj.data.materials.append(mat)
        for polygon, source_face in zip(obj.data.polygons, keep):
            polygon.use_smooth = mesh.polygons[source_face].use_smooth
            polygon.material_index = mesh.polygons[source_face].material_index
        for edge in obj.data.edges:
            edge.use_edge_sharp = original_sharp[edge_key(edge)] or edge_key(edge) in requested_sharp
        bpy.context.scene.collection.objects.link(obj)
        assert obj.matrix_world == original.matrix_world
        assert vecs([v.co for v in obj.data.vertices]) == vecs(positions)
        assert [tuple(p.vertices) for p in obj.data.polygons] == [faces[i] for i in keep]
        assert sorted(tuple(sorted(e.vertices)) for e in obj.data.edges) == sorted(tuple(sorted(e)) for e in edges)
        return obj

    phase = 'derive-targeted-sharp-flags-from-zero-baseline'
    baseline = clone('FAN DIAGNOSTIC / zero-removed baseline', set())
    baseline.data.normals_split_custom_set(provided)
    baseline_normals = [n.vector.copy() for n in baseline.data.corner_normals]
    assert vecs(baseline_normals) == ZERO_PRIOR['readbackBeforeField'], 'Same baseline did not reproduce'
    bad_corners = [i for i, value in enumerate(errors(baseline_normals, provided)) if value > .002]
    bad_vertices = sorted({baseline.data.loops[i].vertex_index for i in bad_corners})
    assert bad_corners == [134, 136, 432, 435, 439, 441] and bad_vertices == [38]
    targeted = {edge_key(edge) for edge in baseline.data.edges if any(v in bad_vertices for v in edge.vertices)}
    all_edges = set(original_sharp)
    save('targeted-selection.json', {'thresholdUnchanged': .002, 'baselineSourceNormalReadback': stats(baseline_normals, provided),
        'retainedCornerIndices': bad_corners, 'originalSourceLoopIndices': [source_loops[i] for i in bad_corners],
        'sourceVertexIndices': bad_vertices, 'requestedIncidentEdges': sorted(targeted),
        'selectionRule': 'every existing edge incident to a baseline source-normal mismatch vertex; one nonadaptive pass'})
    variants = [('zero-removed-baseline', baseline, set()),

    function = fields.field_for_role('l14', 'hanging-leaf', 0, P)

    records = []

        finally:
            sys.setprofile(None)
        assert all(key in state for key in ['expected', 'actual', 'matrices', 'old_normals', 'residual']), 'Failure before normal comparison'
        desired = [(state['matrices'][loop.vertex_index] @ normal).normalized() for loop, normal in zip(m.loops, provided)]
        actual = [n.vector.copy() for n in m.corner_normals]
        desired_stats = stats(actual, desired)
        phase = name + ':optional-export-readback'
        output_export = export_and_check(obj, desired, name + '-output') if method_error is None and input_stats['maxL2'] <= .002 and desired_stats['maxL2'] <= .002 else None
        flags_after_field = sharp(m)
        changed = sorted(key for key in original_sharp if original_sharp[key] != flags_after_source[key])
        record = {'variant': name, 'sourceId': {'ownerId': 'l14', 'sourceNodeIndex': 2239, 'forestRole': 'hanging-leaf', 'ordinal': 0},
