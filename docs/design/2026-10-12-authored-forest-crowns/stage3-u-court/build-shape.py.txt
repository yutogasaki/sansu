def save(name,data):
    (OUT/name).write_text(json.dumps(data,indent=2)+'\n')
def volume(obj):
    if obj.name not in VOLUMES:VOLUMES[obj.name]=Volume64(obj,BI)
    return VOLUMES[obj.name]
def role(owner,kind,ordinal=0):
    found=[o for o in ROLES if o['ownerId']==owner and o['forestRole']==kind and o['forestOrdinal']==ordinal]
    assert len(found)==1,(owner,kind,ordinal);return found[0]
def mapped(owner,kind,ordinal,point):
    return Vector(fields.field_for_role(owner,kind,ordinal,PROPOSAL)(point))
def connect(left_obj,right_obj,left_guess,right_guess,radius,max_span):
    assert left_obj['ownerId']==right_obj['ownerId']
    left,right=volume(left_obj),volume(right_obj)
    row={'left':left_obj.name,'right':right_obj.name,'ownerId':left_obj['ownerId'],
         'leftGuess':list(left_guess),'rightGuess':list(right_guess),'ballRadiusWorld':radius*.6,
         'status':'unresolved','clearanceApproximate':True}
    JOINTS.append(row)
    try:
        overlap=joints.overlap_ball(left,right,[left_guess,right_guess],radius*.6)
        if overlap:
            row.update(status='measured-overlap-under-inherited-tolerances',overlap=overlap)
        else:
            assert len(CREATED)<31,'Original derived-bough cap reached'
            material=(right_obj if right_obj['forestRole'] in ('main-branch','hanging-stem') else left_obj).data.materials[0]
            obj,measurement=junctions.connector(left,right,left_guess,right_guess,radius,max_span,
                  f"stage3-junction-{len(JOINTS):02d}-{left_obj['ownerId']}",left_obj['ownerId'],material,
                  JOINT_COLLECTION,B,STUDY['sourceSha256'])
            CREATED.append(obj)
            row.update(status='derived-closed-bough-approximate-endpoint-measurements',measurement=measurement)
    except BaseException as error:
        row['error']=repr(error);raise
    finally:save('primary-junctions.json',JOINTS)

def parity():
    assert fixed_before=={o.name:native.fixed_digest(o) for o in FIXED}
    assert retained_before=={o.name:native.fixed_digest(o) for o in RETAINED}
    assert context_before=={o.name:native.fixed_digest(o) for o in CONTEXT if o.type=='MESH'}
    assert topology_before=={o.name:[list(p.vertices) for p in o.data.polygons] for o in ROLES}
    assert material_before=={m.name:checks.material_digest(m) for m in MATERIALS}
    assert slots_before=={o.name:[m.name for m in o.data.materials] for o in ROLES}
    assert INPUTS=={name:sha(Path(name)) for name in INPUTS}

try:

    JOINT_COLLECTION=bpy.data.collections.new('stage3 / same-owner visible boughs')
