"""One separately authorized build or view. Never launches both implicitly."""
import os
assert os.environ.get('SANSU_FOREST_STAGE3_APPROVED')=='1','Await source window'
import argparse
import hashlib
import json
import signal
import subprocess
import time
from datetime import datetime,timezone
from pathlib import Path
HERE=Path(__file__).resolve().parent
parser=argparse.ArgumentParser();parser.add_argument('phase',choices=['build','view']);args=parser.parse_args()
OUT=HERE/('execution-'+args.phase+'-01');assert not OUT.exists(),'Preserve first attempt';OUT.mkdir()
def snapshot():
    ready=HERE/'readiness.json';sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
    assert sha(ready)==(HERE/'readiness.sha256').read_text().strip(),'Preparation seal mismatch'
    record=json.loads(ready.read_text());actual={name:sha(HERE/name) for name in record['files']}
    assert actual==record['files'],'Prepared script/parameters changed'
    actual['readiness.json']=sha(ready);actual['readiness.sha256']=sha(HERE/'readiness.sha256')
    return actual
bound_before=snapshot()
script,timeout=('build-shape.py',300) if args.phase=='build' else ('render-whole.py',120)
command=['/Applications/Blender.app/Contents/MacOS/Blender','-b','--threads','2','--python-exit-code','1','--python',str(HERE/script)]
started=datetime.now(timezone.utc).isoformat();start=time.monotonic();peak=0;stop=None
with (OUT/'execution.log.txt').open('xb') as log:
    p=subprocess.Popen(command,stdout=log,stderr=subprocess.STDOUT,start_new_session=True,
                       env=dict(os.environ,OMP_NUM_THREADS='2',OPENBLAS_NUM_THREADS='1'))
    try:
        while p.poll() is None:
            rss=subprocess.run(['ps','-o','rss=','-p',str(p.pid)],capture_output=True,text=True,check=False).stdout.strip()
            if rss:peak=max(peak,int(rss)*1024)
            if peak>3*1024**3:stop='observed RSS 3 GiB limit';break
            if time.monotonic()-start>timeout:stop='wall limit';break
            time.sleep(.25)
    finally:
        if p.poll() is None:
            os.killpg(p.pid,signal.SIGTERM)
            try:p.wait(timeout=10)
            except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
code=(125 if 'RSS' in stop else 124) if stop else p.returncode
binding_error=None;bound_after=None
try:bound_after=snapshot();assert bound_before==bound_after
except Exception as error:binding_error=repr(error);code=126
record={'phase':args.phase,'command':command,'startedAtUtc':started,'finishedAtUtc':datetime.now(timezone.utc).isoformat(),
        'elapsedSeconds':time.monotonic()-start,'exitCode':code,'ownedPid':p.pid,'peakObservedRssBytes':peak,
        'rssSampleSeconds':.25,'rssGuardBytes':3*1024**3,'timeoutSeconds':timeout,'effectiveThreads':2,
        'stopReason':stop,'boundInputsBefore':bound_before,'boundInputsAfter':bound_after,'bindingError':binding_error,
        'scope':'source-only HOLD; no runtime or performance acceptance'}
(OUT/'result.json').write_text(json.dumps(record,indent=2)+'\n');print(json.dumps(record),flush=True)
raise SystemExit(code)
