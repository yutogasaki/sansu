def glb_json(path):
    with path.open('rb') as stream:
        assert stream.read(4) == b'glTF'
        version, total = struct.unpack('<II', stream.read(8))
        length, kind = struct.unpack('<II', stream.read(8))
        assert version == 2 and kind == 0x4e4f534a and total == path.stat().st_size
        return json.loads(stream.read(length))


def material_spec(item):
    pbr = item.get('pbrMetallicRoughness', {})
    def no_textures(value):
        if isinstance(value, dict):
            assert not any('texture' in key.lower() for key in value), 'Do not mock or drop textures'
            for child in value.values(): no_textures(child)
        elif isinstance(value, list):
            for child in value: no_textures(child)
    no_textures(item)
    return {'baseColor': pbr.get('baseColorFactor', [1, 1, 1, 1]),
            'roughness': pbr.get('roughnessFactor', 1), 'metalness': pbr.get('metallicFactor', 1),
            'doubleSided': item.get('doubleSided', False), 'alphaMode': item.get('alphaMode', 'OPAQUE'),
            'emissive': item.get('emissiveFactor', [0, 0, 0])}


def close_values(a, b):
    if isinstance(a, dict): return a.keys() == b.keys() and all(close_values(a[k], b[k]) for k in a)
    if isinstance(a, list): return len(a) == len(b) and all(close_values(x, y) for x, y in zip(a, b))
    if isinstance(a, (int, float)) and not isinstance(a, bool): return abs(a - b) <= 2e-6
    return a == b


def exported_roles(document):
    result = {}
    for node in document.get('nodes', []):
        extra = node.get('extras', {})
        if 'sourceNodeIndex' not in extra or 'mesh' not in node: continue
        key = (extra['ownerId'], int(extra['sourceNodeIndex']))
        assert key not in result
        primitives = document['meshes'][node['mesh']]['primitives']
        assert all('NORMAL' in primitive['attributes'] for primitive in primitives)
        result[key] = {'role': extra['forestRole'], 'fixed': bool(extra['forestFixed']),
                       'sourceSha256': extra['sourceSha256'],
                       'materials': [material_spec(document['materials'][primitive['material']]) for primitive in primitives]}
    return result


def material_digest(material):
    values = {'diffuse': list(material.diffuse_color), 'roughness': material.roughness,
              'metallic': material.metallic, 'backfaceCulling': material.use_backface_culling,
              'useNodes': material.use_nodes, 'nodes': [], 'links': []}
    if material.node_tree:
        for node in material.node_tree.nodes:
            inputs = []
            for socket in node.inputs:
                if not hasattr(socket, 'default_value'): continue
                value = socket.default_value
                if not isinstance(value, (int, float, bool, str)):
                    try: value = list(value)
                    except TypeError: value = str(value)
                inputs.append([socket.name, value])
            values['nodes'].append([node.name, node.bl_idname, inputs, getattr(getattr(node, 'image', None), 'name', None)])
        values['links'] = sorted([link.from_node.name, link.from_socket.name, link.to_node.name, link.to_socket.name]
                                for link in material.node_tree.links)
    return hashlib.sha256(json.dumps(values, sort_keys=True).encode()).hexdigest()


def role_object(owner, role, ordinal=0):
    found = [obj for obj in ROLE_OBJECTS if obj.get('ownerId') == owner and obj.get('forestRole') == role
             and obj.get('forestOrdinal') == ordinal]
    assert len(found) == 1, (owner, role, ordinal, len(found))
    return found[0]


VOLUMES = {}
def volume(obj):
    if obj.name not in VOLUMES:
        VOLUMES[obj.name] = joints.Volume(obj, BI)
    return VOLUMES[obj.name]


def mapped(owner, role, ordinal, point):
    return Vector(fields.field_for_role(owner, role, ordinal, PROPOSAL)(point))


def original_world(owner, point):
    return native.world_matrix(owner) @ Vector(point)


def checkpoint():

def query_joint(left_obj, right_obj, left_guess, right_guess, radius):
    """Read-only scalar result; identical kernel/arguments for baseline and build.

    Let query errors propagate. Only the baseline recorder catches them as
    diagnostic states; generated geometry keeps its original strict failure.
    """
    assert left_obj['ownerId'] == right_obj['ownerId'], 'Cross-owner structural joins are prohibited'
    left, right = volume(left_obj), volume(right_obj)
    overlap = joints.overlap_ball(left, right, [left_guess, right_guess], radius * .6)
    return {'left': left_obj.name, 'right': right_obj.name, 'ownerId': left_obj['ownerId'],
            'requestedBallRadiusWorld': radius * .6, 'leftClosure': left.closure,
            'rightClosure': right.closure, 'overlap': overlap}


def query_error_status(error):
    message = str(error)
    if message.startswith('Actual volume is not closed/oriented:'):
        return 'closure-invalid'
    if message.startswith(('Ambiguous actual-volume ray parity:', 'Ray parity exceeded bounded intersections:')):
        return 'ray-parity-unresolved'
    return 'query-error'


def compare_baseline(key, after_status):
    baseline = BASELINE_ROWS[key]
    before = baseline['status']
    if after_status not in ('overlap-found', 'overlap-not-found'):
        category = 'after-query-unresolved'
    elif before not in ('overlap-found', 'overlap-not-found'):
        category = 'baseline-query-unresolved'
    elif before == 'overlap-found' and after_status == 'overlap-not-found':
        category = 'overlap-proof-lost-after-deformation'
    elif before == 'overlap-not-found' and after_status == 'overlap-not-found':
        category = 'already-unproven-in-unmodified-imported-input'
    elif before == 'overlap-not-found':
        category = 'overlap-proof-found-after-deformation'
    else:
        category = 'overlap-proof-retained'
    return {'baselineKey': key, 'baselineStatus': before, 'afterStatus': after_status,
            'classification': category,
            'limit': 'Finite no-ball search is not a proof of noncontact; baseline is after Blender import.'}


def leaf_joint_spec(anchor, deformed):
    ordinal = anchor['k'] * 5 + anchor['j']
    stem = role_object('l14', 'hanging-stem', anchor['k'])
    leaf = role_object('l14', 'hanging-leaf', ordinal)
    original = original_world('l14', anchor['originGltf'])
    at = mapped('l14', 'hanging-leaf', ordinal, original) if deformed else original
    metadata = {'key': f"l14:{int(stem['sourceNodeIndex'])}:{int(leaf['sourceNodeIndex'])}",
                'ownerId': 'l14', 'stemSourceNode': int(stem['sourceNodeIndex']),
                'leafSourceNode': int(leaf['sourceNodeIndex']), 'k': anchor['k'], 'j': anchor['j'],
                'sourceAnchorGltf': anchor['originGltf'], 'originalAnchorWorld': list(original),
                'queryAnchorWorld': list(at), 'anchorMapping': 'reviewed-field' if deformed else 'identity',
                'radiusArgumentWorld': .0008, 'actualOverlapBallRadiusWorld': .0008 * .6,
                'maxSpanArgumentWorld': .04, 'repairAllowed': False,
                'positionWeldDecimalsWorld': 7, 'rayAdvanceWorld': 1e-7}
    return stem, leaf, at, metadata


def record_leaf_baseline():
    """Query all 70 original pairs before any deformation; retain scalars only."""
    global ACTIVE_JOINT_CONTEXT
    originals_before = {obj.name: native.fixed_digest(obj) for obj in ROLE_OBJECTS}
    metadata = {'sourceSha256': STUDY['sourceSha256'],
                'extractionSha256': sha(EXTRACT / 'role-separated-originals.glb'),
                'queryImplementation': {'builderSha256': sha(Path(__file__)), 'jointsSha256': sha(HERE / 'joints.py')},
                'scope': 'Unmodified imported input, no connector generation; not an independent raw-GLB mesh audit',
                'ownerBinding': {key: BINDINGS['l14'][key] for key in
                                 ['record', 'sourcePivot', 'sourceScale', 'ageScale',
                                  'authoredTreeMatrixWorld', 'sourceToAuthoredMatrix']}}
    def flush():
        (OUTPUT / 'baseline-joints.json').write_text(json.dumps({**metadata, 'rows': BASELINE_ROWS}, indent=2) + '\n')
    try:
        for anchor in STUDY['anchors']['willow']['leafOriginAnchors']:
            stem, leaf, at, spec = leaf_joint_spec(anchor, deformed=False)
            ACTIVE_JOINT_CONTEXT = spec
            try:
                result = query_joint(stem, leaf, at, at, radius=.0008)
                row = {**spec, 'status': 'overlap-found' if result['overlap'] else 'overlap-not-found',
                       'measurement': result}
            except Exception as error:
                row = {**spec, 'status': query_error_status(error),
                       'errorClass': type(error).__name__, 'error': str(error)}
            assert spec['key'] not in BASELINE_ROWS
            BASELINE_ROWS[spec['key']] = row
            if len(BASELINE_ROWS) % 10 == 0:
                flush()
        assert len(BASELINE_ROWS) == 70
        assert originals_before == {obj.name: native.fixed_digest(obj) for obj in ROLE_OBJECTS}
        metadata['originalRoleDigestsUnchanged'] = True
    finally:
        metadata['cachedVolumesBeforeClear'] = len(VOLUMES)
        VOLUMES.clear()
        gc.collect()
        metadata['cachedVolumesAfterClear'] = len(VOLUMES)
        ACTIVE_JOINT_CONTEXT = None
        flush()
    assert not VOLUMES
    print('STAGE2_BASELINE_RECORDED', len(BASELINE_ROWS), 'cache-cleared', flush=True)


def connect(left_obj, right_obj, left_guess, right_guess, radius, max_span, repair=True, baseline_spec=None):
    global ACTIVE_JOINT_CONTEXT
    ACTIVE_JOINT_CONTEXT = baseline_spec
    measurement = query_joint(left_obj, right_obj, left_guess, right_guess, radius)
    overlap = measurement.pop('overlap')
    row = measurement
    if baseline_spec:
        row['sourceJoint'] = baseline_spec
        row['baselineComparison'] = compare_baseline(baseline_spec['key'],
                                                    'overlap-found' if overlap else 'overlap-not-found')
    name = f"joint-{len(JOINTS):03d}-{left_obj['ownerId']}-{right_obj.name}"
    if overlap:
        row.update({'generated': False, 'status': 'actual-positive-volume-overlap-ball', 'overlap': overlap})
    elif repair:
        assert len(CREATED) < 31, 'Bounded native joint repairs exhausted; preserve the first failure'
        left, right = volume(left_obj), volume(right_obj)
        obj, proof = joints.connector(left, right, left_guess, right_guess, radius, max_span, name,
                                      left_obj['ownerId'], right_obj.data.materials[0] if right_obj['forestRole'] in ('main-branch', 'hanging-stem')
                                      else left_obj.data.materials[0], JOINT_COLLECTION, B, STUDY['sourceSha256'])
        CREATED.append(obj)
        row.update({'status': 'closed-native-connector-with-both-cap-disks-embedded', 'connector': proof})
    else:
        row['status'] = 'HOLD-no-positive-volume-attachment-proven; no automatic leaf decoration repair'
        JOINTS.append(row); checkpoint()
        raise ValueError(f'Native hanging leaf joint requires inspection: {name}')
    JOINTS.append(row); checkpoint()
    ACTIVE_JOINT_CONTEXT = None
