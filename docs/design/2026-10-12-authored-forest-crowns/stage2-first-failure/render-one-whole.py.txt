"""Exactly one whole source view after a successful source builder."""
import os
assert os.environ.get('SANSU_FOREST_STAGE2_APPROVED') == '1', 'Await parent execution window'
import bpy
import hashlib
import json
import time
import traceback
from pathlib import Path

ROOT = Path(__file__).resolve().parent / 'attempt-01'
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
assert not (ROOT / 'first-failure.json').exists(), 'Do not render failed source as a candidate'
manifest = json.loads((ROOT / 'manifest.json').read_text())
for name, digest in manifest['files'].items(): assert sha(ROOT / name) == digest
inputs = {name: sha(ROOT / name) for name in ['study.blend', 'study.glb', 'owned-tree-components.glb', 'manifest.json']}
output = ROOT / 'source-whole.png'
assert not output.exists(), 'Preserve first image'
start = time.monotonic()
try:
    bpy.ops.wm.open_mainfile(filepath=str(ROOT / 'study.blend'))
    scene = bpy.context.scene
    assert scene['candidate'] == manifest['candidate'] and 'whole' in scene.camera.name
    scene.render.engine = 'CYCLES'; scene.cycles.device = 'CPU'; scene.cycles.samples = 16
    scene.cycles.use_denoising = True; scene.render.threads_mode = 'FIXED'; scene.render.threads = 2
    assert scene.render.resolution_x == 1280 and scene.render.resolution_percentage == 100
    scene.render.image_settings.file_format = 'PNG'; scene.render.filepath = str(output)
    bpy.ops.render.render(write_still=True)
    assert inputs == {name: sha(ROOT / name) for name in inputs}
    result = {'candidate': scene['candidate'], 'evidenceType': 'offline source whole render, NOT actual app',
              'inputsBefore': inputs, 'inputsAfter': inputs, 'imageSha256': sha(output),
              'cameraReferenceSha256': scene.camera['evidenceSha256'], 'fixedReferenceBuild': 'authored-shoulder-c1',
              'elapsedSeconds': time.monotonic() - start, 'settings': {'engine': 'Cycles CPU', 'samples': 16, 'threads': 2,
              'resolution': [scene.render.resolution_x, scene.render.resolution_y], 'viewTransform': scene.view_settings.view_transform},
              'limitations': scene['renderLimitations'], 'runtimeIntegrated': False, 'independentObservers': 0}
    (ROOT / 'render-manifest.json').write_text(json.dumps(result, indent=2) + '\n')
    print('STAGE2_SOURCE_WHOLE_RENDERED', result['imageSha256'], flush=True)
except BaseException as error:
    with (ROOT / 'first-render-failure.json').open('x') as stream:
        json.dump({'error': repr(error), 'traceback': traceback.format_exc(), 'inputsBefore': inputs,
                   'elapsedSeconds': time.monotonic() - start, 'runtimeIntegrated': False}, stream, indent=2)
    raise
