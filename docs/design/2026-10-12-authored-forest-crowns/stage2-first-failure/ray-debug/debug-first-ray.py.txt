def f32_info(value):
    bits = struct.unpack('<I', struct.pack('<f', value))[0]
    if value == 0:
        above, below = 1, 0x80000001
    elif value > 0:
        above, below = bits + 1, bits - 1
    else:
        above, below = bits - 1, bits + 1
    unpack = lambda b: struct.unpack('<f', struct.pack('<I', b))[0]
    return {'bitsHex': f'{bits:08x}', 'ulpUp': unpack(above) - value,
            'ulpDown': value - unpack(below)}

class ObservedTree:
    """Return original BVH results; all extra arithmetic is diagnostic only."""
    def __init__(self, original):
        self.original = original
    def find_nearest(self, *args):
        return self.original.find_nearest(*args)
    def ray_cast(self, origin, direction, distance):
        result = self.original.ray_cast(origin, direction, distance)
        hit, normal, triangle, hit_distance = result
        entry = {'queryIndex': active['index'], 'call': len(rays),
                 'origin': list(origin), 'direction': list(direction),
                 'maxDistance': distance, 'hit': list(hit) if hit is not None else None,
                 'normal': list(normal) if normal is not None else None,
                 'triangle': triangle, 'distance': hit_distance}
        previous = rays[-1] if rays else None
        if previous and previous['queryIndex'] == active['index'] and previous['direction'] == entry['direction']:
            entry['matchesPreviousNextOrigin'] = entry['origin'] == previous.get('nextOrigin')
            entry['sameHitAndTriangle'] = entry['hit'] == previous['hit'] and triangle == previous['triangle']
        if hit is not None:
            # Exactly the original mathutils expression; never returned to the kernel.
            next_origin = hit + direction * 1e-7
            delta = [float(next_origin[i]) - float(hit[i]) for i in range(3)]
            entry.update({
                'nextOrigin': list(next_origin), 'advanceFromHit': delta,
                'advanceFromHitLength': math.sqrt(sum(v * v for v in delta)),
                'originToNext': [float(next_origin[i]) - float(origin[i]) for i in range(3)],
                'doubleIntendedNext': [float(hit[i]) + float(direction[i]) * 1e-7 for i in range(3)],
                'doubleIntendedDelta': [float(direction[i]) * 1e-7 for i in range(3)],
                'hitFloat32': [f32_info(float(v)) for v in hit],
                'nextFloat32': [f32_info(float(v)) for v in next_origin],
            })
        rays.append(entry)
        return result
