P=json.loads((HERE/'proposal.json').read_text()); rows=[]

 for y in [2.2,2.8,3.174249989,3.7]:

 owner=P['owners'][owner_id];root=owner['branchRootPin']['anchorWorld'];stop=root[1]+owner['branchRootPin']['fixedBandWorld']

  tip=pair['anchorWorld'];target=pair['targetTipWorld'];L=tip[1]-stop

  for y in [root[1]-.1,root[1],stop-1e-10,stop]:
   for dx,dz in [(-.08,-.06),(0,0),(.07,.03)]:
    p=(root[0]+dx,y,root[2]+dz);assert branch(p)==p
  # Triangular map: det = radialScale(y)^2 * dY/dy. radialScale >= 1.
  # s'=30 t²(1-t)², bell'=192 t²(1-t)²(1-2t), maximum t²(1-t)²=1/16.
  dy=target[1]-tip[1];by=pair['bowOffsetWorld'][1]
  determinant_lower=1+min(0.,30*dy-192*abs(by))/(16*L)
  assert pair['distalWidthScale']>=1 and pair['bodyWidthBulge']>=0
  assert determinant_lower>P['guards']['minimumSampledJacobianDeterminant']
  samples=[]
  for j in range(9):
   t=j/8;y=stop+t*L;source=fields.guide(y,owner,pair)
   a=branch(source);b=crown(source);assert a==b
   samples.append({'t':t,'sourceGuide':source,'mappedGuide':a})
  # No derivative jump at the fixed-band or distal crown boundary.
  for y in [stop,tip[1]]:
   p=(tip[0]+.05,y,tip[2]-.03);eps=1e-6
   lower=branch((p[0],y-eps,p[2]));center=branch(p);upper=branch((p[0],y+eps,p[2]))

   assert err<.002
  rows.append({'ownerId':owner_id,'ordinal':i,'sourceBranchNode':pair['sourceBranchNode'],
