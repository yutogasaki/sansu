"""Pure authored-Y-up fields. Original topology, source matrices and roles retained.

A branch and its leaf crown use exactly the same field. This preserves shared
coordinates, not an unmeasured source seam. Root-height band is exact identity.
The recipe guide is only a radial reference: the original mesh residual,
including its existing curve and taper, remains present.
"""
import math


def add(a,b): return tuple(x+y for x,y in zip(a,b))
def sub(a,b): return tuple(x-y for x,y in zip(a,b))
def mul(a,s): return tuple(x*s for x in a)
def length(a): return math.sqrt(sum(x*x for x in a))
def smooth(t):
    t=max(0.,min(1.,t));return t*t*(3.-2.*t)
def smoother(t):
    t=max(0.,min(1.,t));return t*t*t*(10.+t*(-15.+6.*t))
def bell(t):
    t=max(0.,min(1.,t));return 64.*t*t*t*(1.-t)**3

def willow(point,owner):
    # Exact stage2 field, including the original fixed tip core.
    weighted=[0.,0.,0.]; total=0.
    for hint in owner['lobeHints']:
        delta=sub(point,hint['anchorWorld']); width=hint['influenceRadiusXZWorld']
        weight=math.exp(-(delta[0]**2+delta[2]**2)/(width*width))
        total+=weight
        for i in range(3):weighted[i]+=weight*(delta[i]*(hint['stretch'][i]-1.)+hint['shiftWorld'][i])
    if total<1e-12:raise ValueError('Outside common native willow influence domain')
    core=owner['fixedTipCore']; w=smooth((length(sub(point,core['anchorWorld']))-core['radiusWorld'])/core['falloffWorld'])
    return add(point,tuple(value/total*w for value in weighted))

def guide(point_y,owner,pair):
    """C2 radial reference through original recipe root/mid/tip.

    This does NOT reconstruct Blender AUTO handles or replace source geometry.
    End slopes zero keep the above-tip crown field continuously differentiable.
    """
    root=owner['branchRootPin']['anchorWorld'];mid=pair['guideMiddleWorld'];tip=pair['anchorWorld']
    if point_y<=mid[1]:a,b=root,mid
    else:a,b=mid,tip
    t=smoother((point_y-a[1])/(b[1]-a[1]))
    return (a[0]+(b[0]-a[0])*t,point_y,a[2]+(b[2]-a[2])*t)

def birch(point,owner,ordinal):
    pin=owner['branchRootPin']; pair=owner['branchCrownPairs'][ordinal]
    root=pin['anchorWorld']; tip=pair['anchorWorld']; stop=root[1]+pin['fixedBandWorld']
    if point[1]<=stop:return tuple(point)
    t=(point[1]-stop)/(tip[1]-stop)
    w=smoother(t); bulge=bell(t)
    offset=add(mul(sub(pair['targetTipWorld'],tip),w),mul(pair['bowOffsetWorld'],bulge))
    radial=1.+w*(pair['distalWidthScale']-1.)+bulge*pair['bodyWidthBulge']
    axis=guide(point[1],owner,pair)
    return (point[0]+offset[0]+(radial-1.)*(point[0]-axis[0]),
            point[1]+offset[1],
            point[2]+offset[2]+(radial-1.)*(point[2]-axis[2]))

def field_for_role(owner_id,role,ordinal,proposal):
    owner=proposal['owners'][owner_id]
    if owner_id=='l14' and role in ('crown-lobe','hanging-stem','hanging-leaf'):
        return lambda point:willow(point,owner)
    if owner_id in ('l15','l29') and role in ('crown-lobe','main-branch'):
        return lambda point:birch(point,owner,ordinal)
    if owner_id=='l28' and role=='crown-tier':return lambda point:tuple(point)
    raise ValueError('Fixed/unknown role: '+str((owner_id,role,ordinal)))
