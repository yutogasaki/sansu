"""Bounded external floor diagnosis; fixed app candidate is read only."""
import argparse, hashlib, json, os, signal, subprocess, time
from datetime import datetime, timezone
from pathlib import Path
HERE = Path(__file__).resolve().parent
DRIVER = HERE / 'diagnose-floor.mjs'
EXPECTED = 'befd9cf6f7aeb91a2b27e27c96c416d19954fb24cd5303e3cc753619451d6fc1'
p = argparse.ArgumentParser()
p.add_argument('--start', action='store_true')
p.add_argument('--mode', choices=['shadow-off', 'unlit'], default='shadow-off')
p.add_argument('--shadow-report')
p.add_argument('--line-review')
a = p.parse_args()
assert hashlib.sha256(DRIVER.read_bytes()).hexdigest() == EXPECTED
assert a.start, 'Explicit --start required after all eight fixed c2 stages pass'
if a.mode == 'unlit': assert a.shadow_report and a.line_review
RUN = HERE / (a.mode + '-execution-01')
OUT = HERE / (a.mode + '-01')
assert not RUN.exists() and not OUT.exists(), 'Preserve first attempt'
RUN.mkdir()
cmd = ['node', str(DRIVER), '--start', '--mode', a.mode, '--output', str(OUT)]
if a.mode == 'unlit': cmd += ['--shadow-report', a.shadow_report, '--line-review', a.line_review]
started = datetime.now(timezone.utc).isoformat()
start = time.monotonic(); peak = 0; stop = None
with (RUN / 'execution.log.txt').open('xb') as log:
 proc = subprocess.Popen(cmd, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
 pid = proc.pid
 try:
  while proc.poll() is None:
   rows = subprocess.run(['ps', '-axo', 'pid=,ppid=,rss='], capture_output=True, text=True, check=True).stdout.splitlines()
   table = [tuple(map(int, row.split())) for row in rows if len(row.split()) == 3]
   ids = {pid}
   for _ in range(8): ids.update(child for child, parent, rss in table if parent in ids)
   rss = sum(rss for child, parent, rss in table if child in ids) * 1024
   peak = max(peak, rss)
   if rss > 4 * 1024**3: stop = 'Observed process-tree RSS 4 GiB boundary'
   if time.monotonic() - start > 900: stop = '900 second wall boundary'
   if stop: break
   time.sleep(.5)
 finally:
  if proc.poll() is None:
   os.killpg(pid, signal.SIGTERM)
   try: proc.wait(timeout=10)
   except subprocess.TimeoutExpired: os.killpg(pid, signal.SIGKILL); proc.wait()
code = proc.wait()
record = {'command':cmd, 'startedAtUtc':started, 'finishedAtUtc':datetime.now(timezone.utc).isoformat(),
 'elapsedSeconds':time.monotonic()-start, 'ownedPid':pid, 'ownedProcessGroup':pid, 'exitCode':code,
 'stopReason':stop, 'peakObservedProcessTreeRssBytes':peak, 'rssSampleSeconds':.5,
 'limits':{'seconds':900,'processTreeRssBytes':4*1024**3}, 'driverBeforeSha256':EXPECTED,
 'driverAfterSha256':hashlib.sha256(DRIVER.read_bytes()).hexdigest(),
 'scope':'Actual renderer causal diagnosis only; no performance, art acceptance or source mutation'}
(RUN / 'result.json').write_text(json.dumps(record, indent=2)+'\n')
print(json.dumps(record), flush=True)
raise SystemExit(0 if code == 0 and not stop and record['driverAfterSha256'] == EXPECTED else 1)
